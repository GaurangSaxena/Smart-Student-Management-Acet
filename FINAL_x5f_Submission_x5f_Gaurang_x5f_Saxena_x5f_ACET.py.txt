import os
from difflib import get_close_matches

# Unique File Name - Sirf tumhara hai
FILE_NAME = "Gaurang_ACET_Students_2025.txt"
SECRET_WATERMARK = "GS_ACET_2025_CSE2_GAURANG_ORIGINAL"

def add_student():
    print("\n--- Add New Student ---")
    name = input("Student ka naam: ")
    roll = input("Roll No: ")
    course = input("Course (CSE/IT): ")
    year = input("Year (2nd Year): ")
    # File Handling - Append Mode
    with open(FILE_NAME, "a") as file:
        file.write(f"{name},{roll},{course},{year}\n")
    print(f"\n[SUCCESS] {name} ka data File me save ho gaya!\n")

def view_all():
    print("\n--- All Student Records ---")
    if not os.path.exists(FILE_NAME):
        print("Koi data nahi hai!\n")
        return
    with open(FILE_NAME, "r") as file:
        data = file.read()
        if not data.strip():
            print("File khali hai!\n")
        else:
            print(data)
            print()

def normal_search():
    roll = input("Search ke liye Roll No likho: ")
    found = False
    if os.path.exists(FILE_NAME):
        with open(FILE_NAME, "r") as file:
            for line in file:
                if roll in line:
                    print(f"Mil Gaya: {line.strip()}")
                    found = True
    if not found:
        print("Student nahi mila!\n")

def ai_smart_search():
    print("\n--- AI Smart Search (Spelling galat bhi chalegi) ---")
    search_name = input("Naam likho: ")
    all_names = []
    if os.path.exists(FILE_NAME):
        with open(FILE_NAME, "r") as file:
            for line in file:
                if line.strip():
                    all_names.append(line.split(",")[0])
    if not all_names:
        print("Pehle data add karo!\n")
        return
    matches = get_close_matches(search_name, all_names, n=3, cutoff=0.5)
    if matches:
        print(f"\n[AI] Aap shayad inhe dhoondh rahe ho: {matches}")
        print("AI ne spelling mistake ke baad bhi naam dhoondh liya!\n")
    else:
        print("AI ko bhi koi close match nahi mila!\n")

def show_stats():
    if not os.path.exists(FILE_NAME):
        print("Koi data nahi!\n")
        return
    with open(FILE_NAME, "r") as file:
        lines = file.readlines()
        print(f"\n--- Project Stats ---")
        print(f"Total Students: {len(lines)}")
        print(f"File Location: {FILE_NAME}")
        print(f"File Handling: Use ho raha hai (Data Permanent hai)")
        print(f"Project By: Gaurang Saxena - ACET Aligarh\n")

while True:
    print("==========================================")
    print("  SMART STUDENT MANAGEMENT SYSTEM")
    print("  Gaurang Saxena | ACET Aligarh | AKTU")
    print("==========================================")
    print("1. Add Student (File Handling)")
    print("2. View All Students")
    print("3. Normal Search (Roll No)")
    print("4. AI Smart Search (Name - AI Based)")
    print("5. Show Stats")
    print("6. Exit")
    print("==========================================")
    choice = input("Choice (1-6): ")
    if choice == '1':
        add_student()
    elif choice == '2':
        view_all()
    elif choice == '3':
        normal_search()
    elif choice == '4':
        ai_smart_search()
    elif choice == '5':
        show_stats()
    elif choice == '6':
        print("\nThank You! Project by Gaurang Saxena - ACET Aligarh")
        print(f"Secret: {SECRET_WATERMARK}")
        break
    else:
        print("Galat choice!\n")
